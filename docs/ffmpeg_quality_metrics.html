 304    def get_framerate(input_file: str, ffmpeg_path: str = "ffmpeg") -> float:
 305        """Parse the FPS from the input file.
 306
 307        First parse it from the ffmpeg banner ("<n> fps"). Some container/codec
 308        combinations (e.g. FFV1 in a .nut) do not print an fps token in the
 309        banner (only "tbr"/"tbn"), so fall back to querying ffprobe for the
 310        exact rational frame rate. ffprobe ships with ffmpeg, so this adds no
 311        new dependency.
 312
 313        Args:
 314            input_file (str): Input file path
 315            ffmpeg_path (str, optional): Path to ffmpeg executable. Defaults to "ffmpeg".
 316
 317        Raises:
 318            FfmpegQualityMetricsError: A generic error
 319
 320        Returns:
 321            float: The FPS parsed
 322        """
 323        cmd = [ffmpeg_path, "-nostdin", "-y", "-i", input_file]
 324
 325        output = run_command(cmd, allow_error=True)
 326        pattern = re.compile(r"(\d+(\.\d+)?) fps")
 327        try:
 328            if pattern_ret := pattern.search(str(output)):
 329                match = pattern_ret.groups()[0]
 330                return float(match)
 331        except Exception:
 332            pass
 333
 334        # Fallback: the ffmpeg banner did not contain an "<n> fps" token (this
 335        # happens for some container/codec combinations, e.g. FFV1 in .nut,
 336        # which only report "tbr"/"tbn"). Query ffprobe for the exact rational
 337        # frame rate instead of failing. Prefer r_frame_rate, then
 338        # avg_frame_rate; skip missing/"0/0" values.
 339        ffprobe_path = get_ffprobe_path(ffmpeg_path)
 340        for field in ("r_frame_rate", "avg_frame_rate"):
 341            try:
 342                stdout, _ = run_command(
 343                    [
 344                        ffprobe_path,
 345                        "-loglevel",
 346                        "error",
 347                        "-select_streams",
 348                        "v:0",
 349                        "-show_entries",
 350                        f"stream={field}",
 351                        "-of",
 352                        "csv=p=0",
 353                        input_file,
 354                    ],
 355                    allow_error=True,
 356                )
 357            except Exception:
 358                continue
 359            value = stdout.strip().splitlines()[0].strip() if stdout.strip() else ""
 360            if not value or value == "0/0":
 361                continue
 362            try:
 363                frac = Fraction(value)
 364            except (ValueError, ZeroDivisionError):
 365                continue
 366            if frac > 0:
 367                return float(frac)
 368
 369        raise FfmpegQualityMetricsError(f"could not parse FPS from file {input_file}!")

 391    def probe_input(input_file: str, ffmpeg_path: str = "ffmpeg") -> InputInfo:
 392        """Read the properties of the first video stream of a file with ffprobe.
 393
 394        ffprobe is expected next to the ffmpeg executable, or in the PATH.
 395
 396        Args:
 397            input_file (str): Input file path
 398            ffmpeg_path (str, optional): Path to ffmpeg executable. Defaults to "ffmpeg".
 399
 400        Raises:
 401            FfmpegQualityMetricsError: If the file does not exist, cannot be read, or has no video stream
 402            FileNotFoundError: If ffprobe cannot be found
 403
 404        Returns:
 405            InputInfo: The video stream properties
 406        """
 407        if not os.path.isfile(input_file):
 408            raise FfmpegQualityMetricsError(f"File not found: {input_file}")
 409
 410        cmd = [
 411            get_ffprobe_path(ffmpeg_path),
 412            "-loglevel",
 413            "error",
 414            "-select_streams",
 415            "v:0",
 416            "-show_entries",
 417            "stream=width,height,r_frame_rate,avg_frame_rate,duration,nb_frames"
 418            ":format=duration",
 419            "-of",
 420            "json",
 421            input_file,
 422        ]
 423        try:
 424            stdout, _ = run_command(cmd)
 425            probe_data = json.loads(stdout)
 426        except (RuntimeError, ValueError) as e:
 427            # the last line of ffprobe's error output names the problem
 428            detail = str(e).strip().splitlines()[-1] if str(e).strip() else ""
 429            raise FfmpegQualityMetricsError(
 430                f"Could not read file {input_file} with ffprobe: {detail}"
 431            ) from e
 432
 433        streams = probe_data.get("streams") or []
 434        if not streams:
 435            raise FfmpegQualityMetricsError(f"No video stream found in {input_file}")
 436        stream = streams[0]
 437
 438        framerate = None
 439        for field in ("r_frame_rate", "avg_frame_rate"):
 440            try:
 441                frac = Fraction(stream.get(field, ""))
 442            except (ValueError, ZeroDivisionError):
 443                continue
 444            if frac > 0:
 445                framerate = float(frac)
 446                break
 447
 448        duration = None
 449        for value in (
 450            stream.get("duration"),
 451            probe_data.get("format", {}).get("duration"),
 452        ):
 453            try:
 454                duration = float(value)
 455                break
 456            except (TypeError, ValueError):
 457                continue
 458
 459        try:
 460            nb_frames = int(stream["nb_frames"])
 461        except (KeyError, TypeError, ValueError):
 462            nb_frames = None
 463
 464        return {
 465            "path": input_file,
 466            "width": int(stream.get("width", 0)),
 467            "height": int(stream.get("height", 0)),
 468            "framerate": framerate,
 469            "duration": duration,
 470            "nb_frames": nb_frames,
 471        }

 473    def check_inputs(self) -> Union[Tuple[InputInfo, InputInfo], None]:
 474        """Probe both input files and log a warning for each property that differs.
 475
 476        Compares resolution, frame rate, and duration (or frame count). This is run
 477        automatically before the first calculation.
 478
 479        Raises:
 480            FfmpegQualityMetricsError: If an input cannot be read or has no video stream
 481
 482        Returns:
 483            Tuple[InputInfo, InputInfo] or None: The properties of the reference and distorted file,
 484                or None if ffprobe is not available
 485        """
 486        try:
 487            ref_info = FfmpegQualityMetrics.probe_input(self.ref, self.ffmpeg_path)
 488            dist_info = FfmpegQualityMetrics.probe_input(self.dist, self.ffmpeg_path)
 489        except FileNotFoundError:
 490            logger.warning(
 491                f"ffprobe not found at {get_ffprobe_path(self.ffmpeg_path)}, skipping input checks"
 492            )
 493            return None
 494
 495        logger.debug(f"Reference input: {ref_info}")
 496        logger.debug(f"Distorted input: {dist_info}")
 497
 498        self._check_resolution(ref_info, dist_info)
 499        self._check_framerate(ref_info, dist_info)
 500        self._check_duration(ref_info, dist_info)
 501
 502        self.input_info = (ref_info, dist_info)
 503        return self.input_info

 519    def _check_framerate(self, ref_info: InputInfo, dist_info: InputInfo) -> None:
 520        """
 521        Warn if the frame rates differ and no frame rate is forced.
 522        """
 523        ref_fps = ref_info["framerate"]
 524        dist_fps = dist_info["framerate"]
 525        if ref_fps is None or dist_fps is None or self.framerate is not None:
 526            return
 527        if not math.isclose(ref_fps, dist_fps, abs_tol=1e-3):
 528            logger.warning(
 529                f"Frame rates differ: distorted {dist_fps:g} fps, reference {ref_fps:g} fps. "
 530                "Frames will not line up in time, so the metrics will be inaccurate. "
 531                "Convert both inputs to the same frame rate, or force one with the -r option."
 532            )

 534    def _check_duration(self, ref_info: InputInfo, dist_info: InputInfo) -> None:
 535        """
 536        Warn if the inputs differ in length, after applying dist_delay.
 537        Skipped if num_frames is set.
 538        """
 539        if self.num_frames is not None:
 540            return
 541
 542        ref_frames = ref_info["nb_frames"]
 543        dist_frames = dist_info["nb_frames"]
 544        ref_fps = ref_info["framerate"]
 545        dist_fps = dist_info["framerate"]
 546        # frame counts are only comparable if frames are paired one by one
 547        same_framerate = self.framerate is not None or (
 548            ref_fps is not None
 549            and dist_fps is not None
 550            and math.isclose(ref_fps, dist_fps, abs_tol=1e-3)
 551        )
 552        if (
 553            self.dist_delay == 0
 554            and same_framerate
 555            and ref_frames is not None
 556            and dist_frames is not None
 557        ):
 558            if ref_frames != dist_frames:
 559                longer = "reference" if ref_frames > dist_frames else "distorted"
 560                logger.warning(
 561                    f"Frame counts differ: distorted {dist_frames}, reference {ref_frames}. "
 562                    f"Only the first {min(ref_frames, dist_frames)} frames are compared; "
 563                    f"the remaining frames of the {longer} video are ignored."
 564                )
 565            return
 566
 567        ref_duration = ref_info["duration"]
 568        dist_duration = dist_info["duration"]
 569        if ref_duration is None or dist_duration is None:
 570            return
 571
 572        # dist_delay trims the start of one of the inputs
 573        ref_duration = max(ref_duration - max(self.dist_delay, 0), 0)
 574        dist_duration = max(dist_duration - max(-self.dist_delay, 0), 0)
 575
 576        # tolerate a difference of one frame
 577        known_fps = [
 578            fps
 579            for fps in (self.framerate, ref_info["framerate"], dist_info["framerate"])
 580            if fps
 581        ]
 582        tolerance = 1 / min(known_fps) if known_fps else 0.05
 583
 584        if abs(ref_duration - dist_duration) > tolerance + 1e-6:
 585            longer = "reference" if ref_duration > dist_duration else "distorted"
 586            delay_note = (
 587                f" (after trimming {abs(self.dist_delay):g} s for the distortion delay)"
 588                if self.dist_delay != 0
 589                else ""
 590            )
 591            logger.warning(
 592                f"Durations differ{delay_note}: distorted {dist_duration:.3f} s, "
 593                f"reference {ref_duration:.3f} s. "
 594                f"Only the first {min(ref_duration, dist_duration):.3f} s are compared; "
 595                f"the rest of the {longer} video is ignored."
 596            )

 598    def _parse_start_offset(self, framerate: float) -> Union[str, None]:
 599        """
 600        Parse the start_offset parameter and convert frame numbers to timestamps if needed.
 601
 602        Args:
 603            framerate (float): The framerate to use for frame-to-timestamp conversion
 604
 605        Returns:
 606            Union[str, None]: The timestamp string for ffmpeg's -ss option, or None if no offset
 607        """
 608        if self.start_offset is None:
 609            return None
 610
 611        # Check if it's a frame number (format: "f:100" or "f:100.5")
 612        if self.start_offset.startswith("f:"):
 613            try:
 614                frame_num = float(self.start_offset[2:])
 615                timestamp = frame_num / framerate
 616                return str(timestamp)
 617            except ValueError:
 618                raise FfmpegQualityMetricsError(
 619                    f"Invalid frame number in start_offset: {self.start_offset}"
 620                )
 621
 622        # Otherwise, assume it's a timestamp string (e.g., "00:00:10" or "10.5")
 623        return self.start_offset

 655        """

 658        # convert both inputs to 10 bit if requested, as recommended for VMAF v1 models

 664                ]
 665            )
 666            dist_label += "10b"
 667            ref_label += "10b"
 668
 669        chains.append(
 670            f"[{dist_label}][{ref_label}]{self._get_filter_opts(self.METRIC_TO_FILTER_MAP[metric_name])}"
 671        )
 672        return chains

 674    def calculate(
 675        self,
 676        metrics: Union[List[MetricName], None] = None,
 677        vmaf_options: Union[VmafOptions, None] = None,
 678    ) -> Dict[MetricName, SingleMetricData]:
 679        """Calculate one or more metrics.
 680
 681        Args:
 682            metrics (list, optional): A list of metrics to calculate.
 683                Possible values are ["ssim", "psnr", "vmaf"].
 684                Defaults to ["ssim", "psnr"].
 685            vmaf_options (dict, optional): VMAF-specific options. Uses defaults if not specified.
 686
 687        Raises:
 688            FfmpegQualityMetricsError: In case of an error
 689            e: A generic error
 690
 691        Returns:
 692            dict: A dictionary of per-frame info, with the key being the metric name and the value being a dict of frame numbers ('n') and metric values.
 693        """
 694        if metrics is None:
 695            metrics = ["ssim", "psnr"]
 696        if not metrics:
 697            raise FfmpegQualityMetricsError("No metrics specified!")
 698
 699        # check available metrics
 700        for metric_name in metrics:
 701            filter_name = self.METRIC_TO_FILTER_MAP.get(metric_name, None)
 702            if filter_name not in self.POSSIBLE_FILTERS:
 703                raise FfmpegQualityMetricsError(f"No such metric '{metric_name}'")
 704            if filter_name not in self.available_filters:
 705                raise FfmpegQualityMetricsError(
 706                    f"Your ffmpeg version does not have the filter '{filter_name}'"
 707                )
 708
 709        # probe the inputs once, before running ffmpeg
 710        if not self._inputs_checked:
 711            self.check_inputs()
 712            self._inputs_checked = True
 713
 714        # set VMAF options specifically
 715        if "vmaf" in metrics:
 716            self._check_libvmaf_availability()
 717            self.vmaf_options = self.DEFAULT_VMAF_OPTIONS.copy()
 718            # override with user-supplied options
 719            if vmaf_options:
 720                for key, value in vmaf_options.items():
 721                    if value is not None:
 722                        self.vmaf_options[key] = value  # type: ignore
 723            self._set_vmaf_model_path(self.vmaf_options["model_path"])
 724
 725        # ffmpeg 7.1 or higher: scale2ref filter is deprecated
 726        # input 0: ref, input 1: dist --> swapped for scale filter
 727
 728        # Apply select filter if num_frames is specified
 729        select_filter = ""
 730        if self.num_frames is not None:
 731            select_filter = f"select='lt(n\\,{self.num_frames})',"
 732
 733        # Apply dist_delay by trimming the leading, unmatched portion of one
 734        # stream so the two streams are temporally aligned before comparison.
 735        # libvmaf (and the other metric filters) pair frames sequentially, so a
 736        # PTS-only shift (e.g. -itsoffset) is cancelled by the setpts reset below
 737        # and has no effect. Trimming happens at decode time - no re-encode.
 738        # dist_delay > 0: distorted is delayed against the reference, so the
 739        #   reference has extra leading frames -> trim the reference.
 740        # dist_delay < 0: the distorted stream leads -> trim the distorted.
 741        ref_trim = f"trim=start={self.dist_delay}," if self.dist_delay > 0 else ""
 742        dist_trim = f"trim=start={abs(self.dist_delay)}," if self.dist_delay < 0 else ""
 743
 744        # Align both streams before the reference is split between the scale
 745        # filter and the metric filter. Feeding the untrimmed reference directly
 746        # to scale while separately trimming it for the metric makes ffmpeg
 747        # buffer the entire skipped lead-in on large positive delays. Real
 748        # captures can start tens of seconds into a source, which previously
 749        # grew memory into gigabytes and could be OOM-killed.
 750        filter_chains = [
 751            f"[0]{ref_trim}{select_filter}settb=AVTB,setpts=PTS-STARTPTS[refaligned]",
 752            "[refaligned]split=2[refscale][refpts]",
 753            f"[1]{dist_trim}{select_filter}settb=AVTB,setpts=PTS-STARTPTS[distaligned]",
 754            f"[distaligned][refscale]scale=rw:rh:flags={self.scaling_algorithm}[distpts]",
 755        ]
 756
 757        # generate split filters depending on the number of models
 758        n_splits = len(metrics)
 759        if n_splits > 1:
 760            for source in ["dist", "ref"]:
 761                suffixes = "".join([f"[{source}{n}]" for n in range(1, n_splits + 1)])
 762                filter_chains.extend(
 763                    [
 764                        f"[{source}pts]split={n_splits}{suffixes}",
 765                    ]
 766                )
 767
 768        # special case, only one metric:
 769        if n_splits == 1:
 770            filter_chains.extend(
 771                self._get_metric_filter_chains(metrics[0], "distpts", "refpts")
 772            )
 773        # all other cases:
 774        else:
 775            for n, metric_name in zip(range(1, n_splits + 1), metrics):
 776                filter_chains.extend(
 777                    self._get_metric_filter_chains(metric_name, f"dist{n}", f"ref{n}")
 778                )
 779
 780        # drop results of earlier calls to avoid accumulate metrics
 781        # rebind (instead of clearing) so previously returned results stay intact
 782        for metric_name in metrics:
 783            self.data[metric_name] = []
 784        self.global_stats = {}
 785
 786        try:
 787            output = self._run_ffmpeg_command(filter_chains, desc=", ".join(metrics))
 788            if self.dry_run:
 789                return {}
 790            self._read_temp_files(metrics)
 791            if output:
 792                self._read_ffmpeg_output(output, metrics)
 793            else:
 794                raise FfmpegQualityMetricsError("ffmpeg output is empty!")
 795        except RuntimeError as e:
 796            if "could not initialize feature extractor" in str(e):
 797                raise FfmpegQualityMetricsError(
 798                    "Your ffmpeg build is linked against a libvmaf version that does not support "
 799                    "one of the features required by the chosen VMAF model. "
 800                    "VMAF v1 models (vmaf_v1.0.16 and newer) require libvmaf 3.2.1 or newer. "
 801                    f"Original error:\n{e}"
 802                ) from e
 803            raise
 804        finally:
 805            self._cleanup_temp_files()
 806
 807        # return only those data entries containing values
 808        return {k: v for k, v in self.data.items() if v}

 810    def _get_libvmaf_filter_opts(self) -> str:
 811        """
 812        Returns:
 813
 814            str: A string to use for VMAF in ffmpeg filter chain
 815        """
 816        # we only have one model, and its path parameter is not optional
 817        all_model_params: Dict[str, str] = {
 818            "path": win_vmaf_model_path_check(self.vmaf_model_path)
 819        }
 820
 821        # add further model parameters
 822        for model_param in self.vmaf_options["model_params"]:
 823            key, value = model_param.split("=")
 824            all_model_params[key] = value
 825
 826        all_model_params_str = "\\:".join(
 827            f"{k}={v}" for k, v in all_model_params.items()
 828        )
 829
 830        vmaf_opts: Dict[str, str] = {
 831            "model": all_model_params_str,
 832            "log_path": win_path_check(self.temp_files["libvmaf"]),
 833            "log_fmt": "json",
 834            "n_threads": str(self.vmaf_options["n_threads"]),
 835            "n_subsample": str(self.vmaf_options["n_subsample"]),
 836            # Terminate at the shorter of the two aligned streams. libvmaf is a
 837            # framesync filter whose default (shortest=0, repeatlast=1) repeats
 838            # the last frame of the shorter stream until the longer one ends.
 839            # After dist_delay trims the leading frames to align the starts, the
 840            # reference is typically still longer than the distorted clip (e.g. a
 841            # full-length source vs a short recording); without shortest=1 those
 842            # trailing reference frames are compared against a frozen distorted
 843            # frame, cratering the pooled score. Compare only the overlap.
 844            "shortest": "1",
 845            "repeatlast": "0",
 846        }
 847
 848        if self.vmaf_options["features"]:
 849            features = []
 850            for feature in self.vmaf_options["features"]:
 851                if not feature.startswith("name"):
 852                    feature = f"name={feature}"
 853                features.append(feature.replace(":", "\\:"))
 854            vmaf_opts["feature"] = "|".join(features)
 855
 856        vmaf_opts_string = ":".join(
 857            f"{k}={v}" for k, v in vmaf_opts.items() if v is not None
 858        )
 859
 860        return vmaf_opts_string

 917    def _parse_ffmpeg_metadata_output(
 918        self, ffmpeg_output: str, metric_name: Literal["vif", "msad"]
 919    ) -> None:
 920        """
 921        Parse the filter output written to ffmpeg's metadata output
 922
 923        Args:
 924            ffmpeg_output (str): The output of ffmpeg's stderr
 925            metric_name (Literal["vif", "msad"]): The name of the metric to parse
 926        """
 927        # Example for VIF:
 928        #
 929        # [Parsed_metadata_4 @ 0x7f995cd08640] frame:1    pts:1       pts_time:0.0401x
 930        # [Parsed_metadata_4 @ 0x7f995cd08640] lavfi.vif.scale.0=0.263582
 931        # [Parsed_metadata_4 @ 0x7f995cd08640] lavfi.vif.scale.1=0.560129
 932        # [Parsed_metadata_4 @ 0x7f995cd08640] lavfi.vif.scale.2=0.626596
 933        # [Parsed_metadata_4 @ 0x7f995cd08640] lavfi.vif.scale.3=0.682183
 934        #
 935        # Example for MSAD:
 936        #
 937        # [Parsed_metadata_6 @ 0x10ad04ea0] lavfi.msad.msad.Y=0.029998
 938        # [Parsed_metadata_6 @ 0x10ad04ea0] lavfi.msad.msad.U=0.019501
 939        # [Parsed_metadata_6 @ 0x10ad04ea0] lavfi.msad.msad.V=0.026455
 940        # [Parsed_metadata_6 @ 0x10ad04ea0] lavfi.msad.msad_avg=0.025318
 941
 942        # Each metadata filter instance (e.g. "[Parsed_metadata_4") prints its own
 943        # "frame:" lines, and the output of several instances is interleaved.
 944        # Collect the frames per instance, then keep only the instances that
 945        # printed values for this metric.
 946        frames_by_instance: Dict[str, List[Dict[str, float]]] = {}
 947        metric_instances: List[str] = []
 948        metric_prefix = f"lavfi.{metric_name}."
 949
 950        for line in ffmpeg_output.split("\n"):
 951            line = line.strip()
 952            if not line.startswith("[Parsed_metadata"):
 953                continue
 954
 955            fields = line.split(" ")
 956            if len(fields) < 4:
 957                continue
 958            instance, entry = fields[0], fields[3]
 959
 960            # a new frame appears
 961            if entry.startswith("frame:"):
 962                # the metadata filter counts from 0, all other metrics count from 1
 963                frame_data: Dict[str, float] = {"n": int(entry.split(":")[1]) + 1}
 964                frames_by_instance.setdefault(instance, []).append(frame_data)
 965                continue
 966
 967            # no frame was set, or no info for this metric
 968            if instance not in frames_by_instance or not entry.startswith(
 969                metric_prefix
 970            ):
 971                continue
 972
 973            if instance not in metric_instances:
 974                metric_instances.append(instance)
 975
 976            key, value = entry.split("=")
 977            key = key.replace(metric_prefix, "").replace(".", "_").lower()
 978            frames_by_instance[instance][-1][key] = round(float(value), 3)
 979
 980        for instance in metric_instances:
 981            self.data[metric_name].extend(frames_by_instance[instance])

 996    def _run_ffmpeg_command(
 997        self, filter_chains: List[str], desc: str = ""
 998    ) -> Union[str, None]:
 999        """
1000        Run the ffmpeg command to get the quality metrics.
1001        The filter chains must be specified manually.
1002        'desc' can be a human readable description for the progress bar.
1003
1004        Returns:
1005            Union[str, None]: The output of ffmpeg's stderr
1006        """
1007        if not self.framerate:
1008            ref_framerate, dist_framerate = self._get_framerates()
1009        else:
1010            ref_framerate = self.framerate
1011            dist_framerate = self.framerate
1012
1013        # Parse start_offset
1014        start_offset_timestamp = self._parse_start_offset(ref_framerate)
1015
1016        cmd = [
1017            self.ffmpeg_path,
1018            "-nostdin",
1019            "-nostats",
1020            "-y",
1021            "-threads",
1022            str(self.threads),
1023        ]
1024
1025        # Add -ss before reference input if start_offset is specified
1026        if start_offset_timestamp is not None:
1027            cmd.extend(["-ss", start_offset_timestamp])
1028
1029        # Add -r before -i only if no start_offset (to avoid seeking issues)
1030        # When seeking is used, -r before -i can interfere with frame-accurate seeking
1031        if start_offset_timestamp is None:
1032            cmd.extend(["-r", str(ref_framerate)])
1033
1034        # Note: dist_delay is applied via a trim filter in the filter graph (see
1035        # calculate), not via -itsoffset. A PTS shift here would be cancelled by
1036        # the setpts=PTS-STARTPTS reset and would not align the streams.
1037        cmd.extend(["-i", self.ref])
1038
1039        # Add -ss before distorted input if start_offset is specified
1040        if start_offset_timestamp is not None:
1041            cmd.extend(["-ss", start_offset_timestamp])
1042
1043        # Add -r before -i only if no start_offset
1044        if start_offset_timestamp is None:
1045            cmd.extend(["-r", str(dist_framerate)])
1046
1047        cmd.extend(
1048            [
1049                "-i",
1050                self.dist,
1051                "-filter_complex",
1052                ";".join(filter_chains),
1053                "-an",
1054                "-f",
1055                "null",
1056                NUL,
1057            ]
1058        )
1059
1060        if self.dry_run:
1061            print(quoted_cmd(cmd))
1062            return ""
1063
1064        if self.progress:
1065            logger.debug(quoted_cmd(cmd))
1066            with FfmpegProgress(cmd, self.dry_run) as ff:
1067                with tqdm(total=100, position=1, desc=desc) as pbar:
1068                    for progress in ff.run_command_with_progress():
1069                        pbar.update(progress - pbar.n)
1070                return ff.stderr
1071        else:
1072            _, stderr = run_command(cmd, dry_run=self.dry_run)
1073            return stderr

1185    def get_global_stats(self) -> GlobalStats:
1186        """
1187        Return a dictionary for each calculated metric, with different statstics
1188
1189        Returns:
1190            dict: A dictionary with stats, each key being a metric name and each value being a dictionary with the stats for every submetric. The stats are: 'average', 'median', 'stdev', 'min', 'max'.
1191        """
1192        for metric_name in self.data:
1193            logger.debug(f"Aggregating stats for {metric_name}")
1194            metric_data = self.data[metric_name]
1195            if len(metric_data) == 0:
1196                continue
1197            submetric_keys = [k for k in metric_data[0].keys() if k != "n"]
1198
1199            stats: Dict[str, GlobalStatsData] = {}
1200            for submetric_key in submetric_keys:
1201                values = [float(frame[submetric_key]) for frame in metric_data]
1202                # Filter out non-finite values (inf, -inf, nan) for robust statistics
1203                finite = [v for v in values if math.isfinite(v)]
1204                # Fallback to [0.0] if all values are non-finite to prevent crashes
1205                all_values = finite if finite else [0.0]
1206
1207                stats[submetric_key] = {
1208                    "average": round(float(mean(all_values)), 3),
1209                    "median": round(float(median(all_values)), 3),
1210                    "stdev": round(
1211                        float(pstdev(finite)) if len(finite) > 1 else 0.0, 3
1212                    ),
1213                    "min": round(min(all_values), 3),
1214                    "max": round(max(all_values), 3),
1215                }
1216            self.global_stats[metric_name] = stats
1217
1218        return self.global_stats

726        # input 0: ref, input 1: dist --> swapped for scale filter

734        # stream so the two streams are temporally aligned before comparison.
735        # libvmaf (and the other metric filters) pair frames sequentially, so a
736        # PTS-only shift (e.g. -itsoffset) is cancelled by the setpts reset below
737        # and has no effect. Trimming happens at decode time - no re-encode.
738        # dist_delay > 0: distorted is delayed against the reference, so the
739        #   reference has extra leading frames -> trim the reference.
740        # dist_delay < 0: the distorted stream leads -> trim the distorted.
741        ref_trim = f"trim=start={self.dist_delay}," if self.dist_delay > 0 else ""

743

745        # filter and the metric filter. Feeding the untrimmed reference directly
746        # to scale while separately trimming it for the metric makes ffmpeg
747        # buffer the entire skipped lead-in on large positive delays. Real
748        # captures can start tens of seconds into a source, which previously
749        # grew memory into gigabytes and could be OOM-killed.

754            f"[distaligned][refscale]scale=rw:rh:flags={self.scaling_algorithm}[distpts]",
755        ]

762                filter_chains.extend(
763                    [
764                        f"[{source}pts]split={n_splits}{suffixes}",
765                    ]
766                )
767
768        # special case, only one metric:
769        if n_splits == 1:
770            filter_chains.extend(
771                self._get_metric_filter_chains(metrics[0], "distpts", "refpts")
772            )
773        # all other cases:
774        else:
775            for n, metric_name in zip(range(1, n_splits + 1), metrics):
776                filter_chains.extend(
777                    self._get_metric_filter_chains(metric_name, f"dist{n}", f"ref{n}")
778                )
779
780        # drop results of earlier calls to avoid accumulate metrics
781        # rebind (instead of clearing) so previously returned results stay intact
782        for metric_name in metrics:
783            self.data[metric_name] = []
784        self.global_stats = {}
785
786        try:
787            output = self._run_ffmpeg_command(filter_chains, desc=", ".join(metrics))
788            if self.dry_run:
789                return {}

791            if output:
792                self._read_ffmpeg_output(output, metrics)
793            else:
794                raise FfmpegQualityMetricsError("ffmpeg output is empty!")
795        except RuntimeError as e:
796            if "could not initialize feature extractor" in str(e):
797                raise FfmpegQualityMetricsError(

801                    f"Original error:\n{e}"

805            self._cleanup_temp_files()

808        return {k: v for k, v in self.data.items() if v}
